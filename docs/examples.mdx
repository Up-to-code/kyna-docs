## 1. REST API Client with JSON Processing

```kyna
async fn fetchUser(userId: int): str? {
    const url = "https://api.kyna.dev/v1/users/" + userId;
    try {
        const res = await fetch(url, { timeout: 3000 });
        if (res.ok) {
            const data = res.json();
            return data["name"];
        }
        return null;
    } catch (err) {
        print("Failed to fetch user:", err.message);
        return null;
    }
}
```

## 2. File Reading & Line Transformation

```kyna
import std.fs;

fn processLogFile(path: str) {
    if (!fs.exists(path)) {
        print("Log file not found:", path);
        return;
    }

    const content = fs.readString(path);
    const lines = strSplit(content, "\n");
    print("Total log entries:", len(lines));

    loop (var i = 0; i < len(lines); i = i + 1) {
        if (strContains(lines[i], "ERROR")) {
            print("ALERT line", i + 1, ":", lines[i]);
        }
    }
}
```

## 3. High-Resolution Benchmarking

```kyna
import std.time;

fn benchmark(iterations: int) {
    const start = time.nowNs();
    var sum = 0;
    loop (var i = 0; i < iterations; i = i + 1) {
        sum = sum + (i % 7);
    }
    const durationNs = time.nowNs() - start;
    print("Processed", iterations, "iterations in", durationNs / 1000000, "ms");
}

benchmark(1000000);
```
