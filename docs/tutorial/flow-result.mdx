# Full Application Flow & Results

This tutorial walks through building a complete, production-ready backend script in Kyna: configuring options, reading files, executing an asynchronous HTTP request, parsing JSON, and handling potential failures cleanly.

---

## Complete Application Code

```kyna
import std.fs;
import std.time;

class ServiceClient {
    private baseUrl: str;
    private apiKey: str;

    public init(baseUrl: str, apiKey: str) {
        self.baseUrl = baseUrl;
        self.apiKey = apiKey;
    }

    public async fn checkHealth(serviceName: str): str {
        const start = time.nowNs();
        const url = self.baseUrl + "/health/" + serviceName;

        try {
            const res = await fetch(url, {
                method: "GET",
                headers: {
                    "Authorization": "Bearer " + self.apiKey,
                    "Accept": "application/json"
                },
                timeout: 5000
            });

            const elapsedMs = (time.nowNs() - start) / 1000000;

            if (res.ok) {
                const data = res.json();
                return "[OK " + elapsedMs + "ms] " + serviceName + " -> Status: " + data["status"];
            } else {
                return "[FAIL " + res.status + "] " + serviceName;
            }
        } catch (err) {
            return "[ERR " + err.code + "] " + serviceName + ": " + err.message;
        }
    }
}

async fn main() {
    print("Initializing Service Client...");
    const client = new ServiceClient("https://api.kyna.dev", "kyna_live_key_992");

    const services = ["auth", "database", "analytics"];
    loop (var i = 0; i < len(services); i = i + 1) {
        const result = await client.checkHealth(services[i]);
        print(result);
    }
}

await main();
```
